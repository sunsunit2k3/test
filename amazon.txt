package com.example.hadoop;

import org.apache.hadoop.io.DoubleWritable;
import org.apache.hadoop.io.Text;
import org.apache.hadoop.mapreduce.Mapper;

import java.io.IOException;
import java.util.List;

public class RevenueMapper extends Mapper<Object, Text, Text, DoubleWritable> {

    private int productIdx = -1;
    private int revenueIdx = -1;

    @Override
    protected void setup(Context context) throws IOException, InterruptedException {

        String headerLine = context.getConfiguration().get("input.header.line");
        if (headerLine == null) return;

        List<String> headers = CSVUtils.parseLine(headerLine);

        for (int i = 0; i < headers.size(); i++) {
            String h = headers.get(i).trim().toLowerCase();
            switch (h) {
                case "product":
                case "product name":
                case "product_name":
                    productIdx = i;
                    break;

                case "revenue":
                    revenueIdx = i;
                    break;
            }
        }
    }

    @Override
    protected void map(Object key, Text value, Context context) throws IOException, InterruptedException {

        String line = value.toString().trim();
        if (line.isEmpty()) return;

        // skip header row
        if (line.toLowerCase().startsWith("date,")) return;

        List<String> cols = CSVUtils.parseLine(line);
        if (cols.size() == 0) return;

        if (productIdx < 0 || revenueIdx < 0) return; // missing field configs
        if (productIdx >= cols.size() || revenueIdx >= cols.size()) return;

        String product = cols.get(productIdx);
        String revenueRaw = cols.get(revenueIdx);

        double revenue = parseRevenue(revenueRaw);

        context.write(new Text(product), new DoubleWritable(revenue));
    }

    private double parseRevenue(String raw) {
        if (raw == null) return 0.0;

        // Remove currency symbols, commas, spaces
        String cleaned = raw.replaceAll("[^0-9.\\-]", "");
        if (cleaned.isEmpty()) return 0.0;

        try {
            return Double.parseDouble(cleaned);
        } catch (Exception e) {
            return 0.0;
        }
    }
}
